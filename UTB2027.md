from math import sqrt

def getOmniwheelMotorValues(
    thrust_x: float,
    thrust_y: float,
    rotation: float,
    half_width: float,
    half_length: float,
) -> tuple[float, float, float, float]:
    """Return FL, FR, BL, BR motor commands as percentages."""

    clamp = lambda value: max(-1.0, min(1.0, value))
    thrust_x = clamp(thrust_x)
    thrust_y = clamp(thrust_y)
    rotation = clamp(rotation)

    wheel_specs = (
        (-1, +1, -half_width, +half_length),  # front-left
        (+1, +1, +half_width, +half_length),  # front-right
        (-1, -1, -half_width, -half_length),  # back-left
        (+1, -1, +half_width, -half_length),  # back-right
    )

    length_scale = max(half_width, half_length)
    if length_scale <= 0:
        raise ValueError("Wheel positions must be nonzero")

    # rotation is scaled so the outer wheel's requested rotational speed
    # is comparable to a full-scale translation request.
    angular_rate = rotation / length_scale
    motor_values = []

    for direction_x, direction_y, wheel_x, wheel_y in wheel_specs:
        drive_x = direction_x / sqrt(2)
        drive_y = direction_y / sqrt(2)

        # Chassis velocity at this wheel: translation plus rotation.
        wheel_vx = thrust_x - angular_rate * wheel_y
        wheel_vy = thrust_y + angular_rate * wheel_x

        motor_values.append(drive_x * wheel_vx + drive_y * wheel_vy)

    if rotation != 0 and all(value == 0 for value in motor_values):
        raise ValueError(
            "These wheel directions and positions provide no rotation authority"
        )

    scale = max(1.0, *(abs(value) for value in motor_values))
    return tuple(100.0 * value / scale for value in motor_values)
